// ==========================================
// 1. TAMIL TRANSLITERATION ENGINE
// ==========================================

object TamilTransliterationEngine {

    private val VOWELS = mapOf(
        'அ' to "a", 'ஆ' to "aa", 'இ' to "i", 'ஈ' to "ee",
        'உ' to "u", 'ஊ' to "oo", 'எ' to "e", 'ஏ' to "ae",
        'ஐ' to "ai", 'ஒ' to "o", 'ஓ' to "oa", 'ஔ' to "au"
    )

    private val CONSONANTS = mapOf(
        'க' to "ka", 'ங' to "nga", 'ச' to "cha", 'ஞ' to "nja",
        'ட' to "ta", 'ண' to "na", 'த' to "tha", 'ந' to "na",
        'ப' to "pa", 'ம' to "ma", 'ய' to "ya", 'ர' to "ra",
        'ல' to "la", 'வ' to "va", 'ழ' to "zha", 'ள' to "la",
        'ற' to "ra", 'ன' to "na",
        'ஜ' to "ja", 'ஷ' to "sha", 'ஸ' to "sa", 'ஹ' to "ha", 'க்ஷ' to "ksha"
    )

    private val VOWEL_SIGNS = mapOf(
        'ா' to "aa", 'ி' to "i", 'ீ' to "ee", 'ு' to "u",
        'ூ' to "oo", 'ெ' to "e", 'ே' to "ae", 'ை' to "ai",
        'ொ' to "o", 'ோ' to "oa", 'ௌ' to "au"
    )

    private const val PULLI = '்'
    private const val AYTHAM = 'ஃ'

    fun transliterate(text: String): String {
        val result = StringBuilder()
        var i = 0
        val len = text.length

        while (i < len) {
            val char = text[i]

            if (char == AYTHAM) {
                result.append("akh")
                i++
                continue
            }

            if (VOWELS.containsKey(char)) {
                result.append(VOWELS[char])
                i++
                continue
            }

            if (CONSONANTS.containsKey(char)) {
                val baseConsonant = CONSONANTS[char]!!
                val nextChar = if (i + 1 < len) text[i + 1] else null

                when {
                    nextChar == PULLI -> {
                        result.append(baseConsonant.dropLast(1))
                        i += 2
                    }
                    nextChar != null && VOWEL_SIGNS.containsKey(nextChar) -> {
                        val vowelSound = VOWEL_SIGNS[nextChar]!!
                        result.append(baseConsonant.dropLast(1)).append(vowelSound)
                        i += 2
                    }
                    else -> {
                        result.append(baseConsonant)
                        i++
                    }
                }
                continue
            }

            result.append(char)
            i++
        }

        return result.toString()
    }
}

// ==========================================
// 2. DATA MODELS & LYRIC PARSER
// ==========================================

data class LrcLine(
    val index: Int = 0,
    val timeMs: Long = 0L,
    val endMs: Long = 0L,
    val tamilText: String = "",
    val transliteration: String = ""
)

object LrcParser {
    private val TIME_TAG_REGEX = Pattern.compile("""\[(\d{1,2}):(\d{2})(?:\.(\d{1,3}))?\]""")

    fun parse(lrcContent: String): List<LrcLine> {
        val list = mutableListOf<Pair<Long, String>>()
        for (line in lrcContent.lines()) {
            val trimmed = line.trim()
            if (trimmed.isEmpty()) continue
            val matcher = TIME_TAG_REGEX.matcher(trimmed)
            var lastEnd = 0
            val timestamps = mutableListOf<Long>()

            while (matcher.find()) {
                val min = (matcher.group(1) ?: "0").toLongOrNull() ?: 0L
                val sec = (matcher.group(2) ?: "0").toLongOrNull() ?: 0L
                val msStr = matcher.group(3) ?: "0"
                val ms = when (msStr.length) {
                    1 -> msStr.toLong() * 100L
                    2 -> msStr.toLong() * 10L
                    else -> msStr.take(3).toLong()
                }

                timestamps.add(min * 60_000L + sec * 1_000L + ms)
                lastEnd = matcher.end()
            }
            if (timestamps.isNotEmpty()) {
                val lyricText = trimmed.substring(lastEnd).trim()
                for (t in timestamps) list.add(Pair(t, lyricText))
            }
        }
        val sorted = list.sortedBy { it.first }
        return sorted.mapIndexed { idx, pair ->
            val nextTime = if (idx + 1 < sorted.size) sorted[idx + 1].first else pair.first + 5000L
            val romanText = TamilTransliterationEngine.transliterate(pair.second)
            
            LrcLine(
                index = idx,
                timeMs = pair.first,
                endMs = nextTime,
                tamilText = pair.second,
                transliteration = romanText
            )
        }
    }
}

// ==========================================
// 3. UI ANIMATION COMPONENT
// ==========================================

@Composable
fun KaraokeProgressiveText(
    text: String,
    progress: Float,
    fontSize: TextUnit = 24.sp,
    fontWeight: FontWeight = FontWeight.Bold,
    highlightColor: Color = Color(0xFFFFD700),
    baseColor: Color = Color.White
) {
    val safeProgress = progress.coerceIn(0f, 1f)

    Box(contentAlignment = Alignment.Center) {
        Text(text = text, fontSize = fontSize, fontWeight = fontWeight, color = baseColor, textAlign = TextAlign.Center)

        Text(
            text = text,
            fontSize = fontSize,
            fontWeight = fontWeight,
            color = highlightColor,
            textAlign = TextAlign.Center,
            modifier = Modifier
                .graphicsLayer { compositingStrategy = CompositingStrategy.Offscreen }
                .drawWithContent {
                    val clipWidth = size.width * safeProgress
                    drawContent()
                    drawRect(
                        color = Color.Transparent,
                        topLeft = Offset(clipWidth, 0f),
                        size = Size(size.width - clipWidth, size.height),
                        blendMode = BlendMode.Clear
                    )
                }
        )
    }
}

// ==========================================
// 4. NAVIGATION & VIEWMODEL
// ==========================================

enum class ScreenState { HOME, STAGE, MAKER_SYNC, LYRIC_EDITOR, RECORDINGS, VIDEOS }

class KaraokeViewModel : ViewModel() {
    private val _currentScreen = MutableStateFlow(ScreenState.HOME)
    val currentScreen: StateFlow<ScreenState> = _currentScreen

    fun navigateTo(screen: ScreenState) {
        _currentScreen.value = screen
    }
}

// ==========================================
// 5. MAIN ACTIVITY & SCREENS
// ==========================================

class MainActivity : ComponentActivity() {
    private val viewModel: KaraokeViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent {
            MaterialTheme(colorScheme = darkColorScheme(background = Color(0xFF121212))) {
                Surface(modifier = Modifier.fillMaxSize(), color = Color(0xFF121212)) {
                    KaraokeAppNavigation(viewModel = viewModel)
                }
            }
        }
    }
}

@Composable
fun KaraokeAppNavigation(viewModel: KaraokeViewModel) {
    val currentScreen by viewModel.currentScreen.collectAsStateWithLifecycle()

    Crossfade(targetState = currentScreen, label = "screen_transition") { screen ->
        when (screen) {
            ScreenState.HOME -> SongLibraryScreen(viewModel)
            ScreenState.STAGE -> KaraokeStageScreen(onBack = { viewModel.navigateTo(ScreenState.HOME) })
            ScreenState.MAKER_SYNC -> SimpleScreen("⏱️ Time Tagging Studio", onBack = { viewModel.navigateTo(ScreenState.HOME) })
            ScreenState.LYRIC_EDITOR -> SimpleScreen("✏️ Offline Lyric Editor", onBack = { viewModel.navigateTo(ScreenState.HOME) })
            ScreenState.RECORDINGS -> SimpleScreen("🎙 Recorded Songs", onBack = { viewModel.navigateTo(ScreenState.HOME) })
            ScreenState.VIDEOS -> SimpleScreen("🎬 Exported Videos", onBack = { viewModel.navigateTo(ScreenState.HOME) })
        }
    }
}

@Composable
fun SongLibraryScreen(viewModel: KaraokeViewModel) {
    Column(
        modifier = Modifier.fillMaxSize().padding(16.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        Text("🎵 Tamil Karaoke App", fontSize = 26.sp, fontWeight = FontWeight.Bold, color = Color(0xFFFFD700))
        Spacer(modifier = Modifier.height(24.dp))
        Button(onClick = { viewModel.navigateTo(ScreenState.STAGE) }) { Text("Start Karaoke Stage") }
        Spacer(modifier = Modifier.height(8.dp))
        Button(onClick = { viewModel.navigateTo(ScreenState.MAKER_SYNC) }) { Text("Time Tagging Studio") }
        Spacer(modifier = Modifier.height(8.dp))
        Button(onClick = { viewModel.navigateTo(ScreenState.LYRIC_EDITOR) }) { Text("Lyric Editor") }
    }
}

@Composable
fun KaraokeStageScreen(onBack: () -> Unit) {
    var progress by remember { mutableFloatStateOf(0.5f) }
    val sampleTamil = "கண்ணே கலைமானே கன்னி மயிலே"
    val transliterated = remember(sampleTamil) { TamilTransliterationEngine.transliterate(sampleTamil) }

    Column(modifier = Modifier.fillMaxSize().padding(16.dp), horizontalAlignment = Alignment.CenterHorizontally) {
        Button(onClick = onBack, modifier = Modifier.align(Alignment.Start)) { Text("⬅ Back") }
        Spacer(modifier = Modifier.height(40.dp))

        Text("🎤 Now Playing", color = Color.Gray, fontSize = 16.sp)
        Spacer(modifier = Modifier.height(30.dp))

        KaraokeProgressiveText(
            text = sampleTamil,
            progress = progress
        )

        Spacer(modifier = Modifier.height(12.dp))

        KaraokeProgressiveText(
            text = transliterated,
            progress = progress,
            fontSize = 18.sp,
            highlightColor = Color(0xFF00EEEE),
            baseColor = Color.LightGray
        )

        Spacer(modifier = Modifier.height(40.dp))
        Text("Simulate Progress:", color = Color.White)
        Slider(value = progress, onValueChange = { progress = it }, valueRange = 0f..1f)
    }
}
